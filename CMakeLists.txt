cmake_minimum_required(VERSION 3.16)
project(vision_training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV REQUIRED)

# This package builds task 3 alone. When merged into the full project,
# the existing task 1 and task 2 sources become separate executables.
add_library(training_common INTERFACE)
target_include_directories(training_common INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(training_common INTERFACE ${OpenCV_LIBS})
add_executable(task3_windmill src/task3_windmill/main.cpp)
target_link_libraries(task3_windmill PRIVATE training_common)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/task1_image/main.cpp")
  add_executable(task1_image src/task1_image/main.cpp)
  target_link_libraries(task1_image PRIVATE training_common)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/task2_fit/main.cpp")
  find_package(Eigen3 REQUIRED NO_MODULE)
  add_executable(task2_fit src/task2_fit/main.cpp)
  target_link_libraries(task2_fit PRIVATE training_common Eigen3::Eigen)
  # Support either the existing Eigen fitter or the separately supplied Ceres fix.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/task2_fit/main.cpp" TASK2_SOURCE_TEXT)
  if(TASK2_SOURCE_TEXT MATCHES "ceres/")
    find_package(Ceres REQUIRED)
    target_link_libraries(task2_fit PRIVATE Ceres::ceres)
  endif()
endif()
